64| async function loadConfigFromDatabase() {

67|         const result = await pool.query('SELECT key, value FROM settings');
68| 
69|         if (result.rows.length === 0) {
70|             console.warn('⚠️ Aucun paramètre trouvé dans la base, utilisation des valeurs par défaut.');
71|             return;
72|         }
73| 
74|         const settings = {};
75|         result.rows.forEach(row => {
76|             settings[row.key] = row.value;
77|         });
78| 
79|         // Mettre à jour les variables globales
80|         if (settings.test_mode !== undefined) TEST_MODE = settings.test_mode === 'true';
81|         if (settings.reports_threshold !== undefined) REPORTS_THRESHOLD = parseInt(settings.reports_threshold) || 2;
82|         if (settings.reports_window_minutes !== undefined) REPORTS_WINDOW_MINUTES = parseInt(settings.reports_window_minutes) || 60;
83|         if (settings.temp_ban_minutes !== undefined) TEMP_BAN_MINUTES = parseInt(settings.temp_ban_minutes) || 10;
84|         if (settings.logs_retention_days !== undefined) LOGS_RETENTION_DAYS = parseInt(settings.logs_retention_days) || 30;
85|         if (settings.reports_retention_days !== undefined) REPORTS_RETENTION_DAYS = parseInt(settings.reports_retention_days) || 60;
86|         if (settings.auto_cleanup_enabled !== undefined) AUTO_CLEANUP_ENABLED = settings.auto_cleanup_enabled === 'true';

89|         if (settings.cleanup_interval_minutes !== undefined) CLEANUP_INTERVAL_MINUTES = parseInt(settings.cleanup_interval_minutes) || 1;
90|         if (settings.dev_mode !== undefined) DEV_MODE = settings.dev_mode === 'true';
