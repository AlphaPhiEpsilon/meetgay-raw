65| // Fonction pour charger la config depuis la base de données
66| async function loadConfigFromDatabase() {
67|     try {
68|         console.log('📂 Chargement des paramètres depuis la base de données...');
69|         const result = await pool.query('SELECT key, value FROM settings');
70| 
71|         if (result.rows.length === 0) {
72|             console.warn('⚠️ Aucun paramètre trouvé dans la base, utilisation des valeurs par défaut.');
73|             return;
74|         }
75| 
76|         const settings = {};
77|         result.rows.forEach(row => {
78|             settings[row.key] = row.value;
79|         });
80| 
81|         // Mettre à jour les variables globales
82|         if (settings.test_mode !== undefined) TEST_MODE = settings.test_mode === 'true';
83|         if (settings.reports_threshold !== undefined) REPORTS_THRESHOLD = parseInt(settings.reports_threshold) || 2;
84|         if (settings.reports_window_minutes !== undefined) REPORTS_WINDOW_MINUTES = parseInt(settings.reports_window_minutes) || 60;
85|         if (settings.temp_ban_minutes !== undefined) TEMP_BAN_MINUTES = parseInt(settings.temp_ban_minutes) || 10;
86|         if (settings.logs_retention_days !== undefined) LOGS_RETENTION_DAYS = parseInt(settings.logs_retention_days) || 30;
87|         if (settings.reports_retention_days !== undefined) REPORTS_RETENTION_DAYS = parseInt(settings.reports_retention_days) || 60;
88|         if (settings.auto_cleanup_enabled !== undefined) AUTO_CLEANUP_ENABLED = settings.auto_cleanup_enabled === 'true';
89|         if (settings.visitor_cleanup_enabled !== undefined) VISITOR_CLEANUP_ENABLED = settings.visitor_cleanup_enabled === 'true';
90|         if (settings.inactivity_minutes !== undefined) INACTIVITY_MINUTES = parseInt(settings.inactivity_minutes) || 2;
91|         if (settings.cleanup_interval_minutes !== undefined) CLEANUP_INTERVAL_MINUTES = parseInt(settings.cleanup_interval_minutes) || 1;
92|         if (settings.dev_mode !== undefined) DEV_MODE = settings.dev_mode === 'true';
93| 
94| 
95|     } catch (err) {
96|         console.error('❌ Erreur lors du chargement des paramètres :', err.message);
97|         console.warn('⚠️ Utilisation des valeurs par défaut (fallback).');
98|     }
