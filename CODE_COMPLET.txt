64| // Fonction pour charger la config depuis la base de données
65| async function loadConfigFromDatabase() {
66|     try {
67|         console.log('📂 Chargement des paramètres depuis la base de données...');
68|         const result = await pool.query('SELECT key, value FROM settings');
69| 
70|         if (result.rows.length === 0) {
71|             console.warn('⚠️ Aucun paramètre trouvé dans la base, utilisation des valeurs par défaut.');
72|             return;
73|         }
74| 
75|         const settings = {};
76|         result.rows.forEach(row => {
77|             settings[row.key] = row.value;
78|         });
79| 
80|         // Mettre à jour les variables globales
81|         if (settings.test_mode !== undefined) TEST_MODE = settings.test_mode === 'true';
82|         if (settings.reports_threshold !== undefined) REPORTS_THRESHOLD = parseInt(settings.reports_threshold) || 2;
83|         if (settings.reports_window_minutes !== undefined) REPORTS_WINDOW_MINUTES = parseInt(settings.reports_window_minutes) || 60;
84|         if (settings.temp_ban_minutes !== undefined) TEMP_BAN_MINUTES = parseInt(settings.temp_ban_minutes) || 10;
85|         if (settings.logs_retention_days !== undefined) LOGS_RETENTION_DAYS = parseInt(settings.logs_retention_days) || 30;
86|         if (settings.reports_retention_days !== undefined) REPORTS_RETENTION_DAYS = parseInt(settings.reports_retention_days) || 60;
87|         if (settings.auto_cleanup_enabled !== undefined) AUTO_CLEANUP_ENABLED = settings.auto_cleanup_enabled === 'true';
88|         if (settings.visitor_cleanup_enabled !== undefined) VISITOR_CLEANUP_ENABLED = settings.visitor_cleanup_enabled === 'true';
89|         if (settings.inactivity_minutes !== undefined) INACTIVITY_MINUTES = parseInt(settings.inactivity_minutes) || 2;
90|         if (settings.cleanup_interval_minutes !== undefined) CLEANUP_INTERVAL_MINUTES = parseInt(settings.cleanup_interval_minutes) || 1;
91|         if (settings.dev_mode !== undefined) DEV_MODE = settings.dev_mode === 'true';
92| 
93| 
94|     } catch (err) {
95|         console.error('❌ Erreur lors du chargement des paramètres :', err.message);
96|         console.warn('⚠️ Utilisation des valeurs par défaut (fallback).');
