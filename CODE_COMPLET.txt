1| // modules/logger.js
2| // ============================================================
3| // Gestionnaire centralisé des logs pour MeetGay
4| // Permet d'activer/désactiver les logs par groupe via l'admin
5| // ============================================================
6| 
7| // ========== DÉFINITION DES GROUPES DE LOGS ==========
8| // Chaque groupe a un nom et un état par défaut
9| const LOG_GROUPS = {
10|     STARTUP: { key: 'log_startup', defaultState: true },
11|     USER_LIFECYCLE: { key: 'log_user_lifecycle', defaultState: true },
12|     MESSAGES: { key: 'log_messages', defaultState: true },
13|     REPORTS: { key: 'log_reports', defaultState: true },
14|     DEBUG: { key: 'log_debug', defaultState: false },
15|     ERRORS: { key: 'log_errors', defaultState: true } // Les erreurs sont toujours utiles
16| };
17| 
18| // ========== ÉTAT DES LOGS EN MÉMOIRE ==========
19| // On charge les valeurs depuis la base, mais on a des fallbacks
20| let logStates = {
21|     log_startup: true,
22|     log_user_lifecycle: true,
23|     log_messages: true,
24|     log_reports: true,
25|     log_debug: false,
26|     log_errors: true
27| };
28| 
29| // ========== FONCTION POUR CHARGER LA CONFIG DEPUIS LA BASE ==========
30| async function loadLogConfig(pool) {
31|     try {
32|         console.log('📋 Chargement de la configuration des logs depuis la base...');
33| 
34|         // On récupère toutes les clés qui commencent par 'log_'
35|         const result = await pool.query(
36|             `SELECT key, value FROM settings WHERE key LIKE 'log_%'`
37|         );
38| 
39|         if (result.rows.length > 0) {
40|             // On met à jour l'état en mémoire
41|             result.rows.forEach(row => {
42|                 if (logStates.hasOwnProperty(row.key)) {
43|                     logStates[row.key] = row.value === 'true';
44|                 }
45|             });
46|             console.log('✅ Configuration des logs chargée avec succès.');
47|             console.log('   État actuel :', logStates);
48|         } else {
49|             console.log('⚠️ Aucune configuration de logs trouvée, utilisation des valeurs par défaut.');
50|         }
51|     } catch (err) {
52|         console.warn('⚠️ Impossible de charger la config des logs :', err.message);
53|         console.warn('   Utilisation des valeurs par défaut.');
54|     }
55| }
56| 
57| // ========== FONCTION PRINCIPALE POUR LOGUER ==========
58| function log(group, ...args) {
59|     // On vérifie si le groupe existe et si il est activé
60|     const groupKey = `log_${group.toLowerCase()}`;
61| 
62|     if (logStates[groupKey]) {
63|         const timestamp = new Date().toISOString();
64|         console.log(`[${timestamp}] [${group.toUpperCase()}]`, ...args);
65|     }
66| }
67| 
68| // ========== FONCTION POUR LES ERREURS ==========
69| // Les erreurs sont toujours loguées, mais on garde une fonction séparée
70| function error(...args) {
71|     // On log toujours les erreurs en console (pour le serveur)
72|     console.error(...args);
73|     // Ici on pourrait ajouter plus tard : envoi par email, stockage en base, etc.
74| }
75| 
76| // ========== FONCTION POUR LES AVERTISSEMENTS ==========
77| function warn(...args) {
78|     // On affiche toujours les avertissements, mais avec un format spécifique
79|     const timestamp = new Date().toISOString();
80|     console.warn(`[${timestamp}] [WARN]`, ...args);
81| }
82| 
83| // ========== FONCTION POUR METTRE À JOUR UN GROUPE ==========
84| // Utile pour l'admin : quand on clique sur un interrupteur
85| async function updateLogGroup(pool, groupKey, enabled) {
86|     try {
87|         // On s'assure que la clé commence par 'log_'
88|         const fullKey = groupKey.startsWith('log_') ? groupKey : `log_${groupKey}`;
89| 
90|         // On met à jour dans la base
91|         await pool.query(
92|             `INSERT INTO settings (key, value) 
93|              VALUES ($1, $2) 
94|              ON CONFLICT (key) DO UPDATE SET value = $2`,
95|             [fullKey, enabled ? 'true' : 'false']
96|         );
97| 
98|         // On met à jour en mémoire
99|         logStates[fullKey] = enabled;
100| 
101|         console.log(`✅ Groupe de logs "${fullKey}" ${enabled ? 'activé' : 'désactivé'}`);
102|         return true;
103|     } catch (err) {
104|         console.error('❌ Erreur lors de la mise à jour du groupe de logs :', err.message);
105|         return false;
106|     }
107| }
108| 
109| // ========== FONCTION POUR RÉCUPÉRER L'ÉTAT D'UN GROUPE ==========
110| function getLogState(groupKey) {
111|     const fullKey = groupKey.startsWith('log_') ? groupKey : `log_${groupKey}`;
112|     return logStates[fullKey] || false;
113| }
