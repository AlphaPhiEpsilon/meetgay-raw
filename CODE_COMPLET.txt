67| async function loadConfigFromDatabase() {

83|         if (settings.test_mode !== undefined) TEST_MODE = settings.test_mode === 'true';
84|         if (settings.reports_threshold !== undefined) REPORTS_THRESHOLD = parseInt(settings.reports_threshold) || 2;
