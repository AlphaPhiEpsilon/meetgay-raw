2| // cookie-consent.js - Gestion de la bannière de cookies
3| // Version sans dépendance externe, utilise ton système Translator

6| const CookieConsent = {
7|     // Nom du cookie de consentement
8|     COOKIE_NAME: 'meetgay_cookie_consent',
9|     // Durée de validité du consentement (6 mois)
10|     EXPIRY_DAYS: 180,
11| 
12|     // État du consentement : null, 'accepted', 'declined'
13|     consent: null,
14| 
15|     // Éléments DOM de la bannière
16|     banner: null,
17| 
18|     /**
19|      * Initialise le système de consentement
20|      */
21|     init: function () {
22|         // 1. Vérifier si le consentement est déjà stocké
23|         this.consent = this.getConsent();
24| 
25|         // 2. Si le consentement n'est pas défini, afficher la bannière
26|         if (!this.consent) {
27|             this.createBanner();
28|         } else if (this.consent === 'accepted') {
29|             // Si déjà accepté, charger les scripts autorisés
30|             this.loadAuthorizedScripts();
31|         }
32|     },
33| 
34|     /**
35|      * Affiche la bannière de cookies (même si déjà cachée)
36|      * Utilisé par le lien "Gérer les cookies" dans le footer
37|      */
38|     showBanner: function () {
39|         // Supprimer l'ancien cookie de consentement
40|         document.cookie = 'meetgay_cookie_consent=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
41|         // Réinitialiser le consentement
42|         this.consent = null;
43|         // Recréer et afficher la bannière
44|         this.createBanner();
45|     },
46| 
47|     /**
48|      * Crée la bannière HTML (avec les traductions)
49|      */
50|     createBanner: function () {
51|         // Si la bannière existe déjà, on ne la recrée pas
52|         if (document.getElementById('cookieConsentBanner')) return;
53| 
54|         // Créer la structure
55|         const banner = document.createElement('div');
56|         banner.id = 'cookieConsentBanner';
57|         banner.className = 'cookie-consent-banner';
58| 
59|         // Utiliser Translator.t() pour les textes
60|         const message = Translator.t('cookie.message');
61|         const acceptText = Translator.t('cookie.accept');
62|         const declineText = Translator.t('cookie.decline');
63|         const settingsText = Translator.t('cookie.settings');
64|         const policyLink = Translator.t('cookie.policy_link');
65| 
66|         banner.innerHTML = `
67|             <p>
68|                 ${message}
69|                 <a href="/cookies-policy.html" class="cookie-policy-link" data-i18n="cookie.policy_link">${policyLink}</a>
70|             </p>
71|             <div class="cookie-buttons">
72|                 <button class="cookie-btn cookie-btn-decline" data-action="decline">${declineText}</button>
73|                 <button class="cookie-btn cookie-btn-accept" data-action="accept">${acceptText}</button>
74|             </div>
75|         `;
76| 
77|         document.body.appendChild(banner);
78|         this.banner = banner;
79| 
80|         // Ajouter les écouteurs d'événements
81|         banner.querySelector('[data-action="accept"]').addEventListener('click', () => {
82|             this.acceptAll();
83|         });
84|         banner.querySelector('[data-action="decline"]').addEventListener('click', () => {
85|             this.declineAll();
86|         });
87| 
88|         // Afficher la bannière après un court délai (pour que le DOM soit prêt)
89|         requestAnimationFrame(() => {
90|             banner.classList.add('show');
91|         });
92|     },
93| 
94|     /**
95|      * Accepte tous les cookies
96|      */
97|     acceptAll: function () {
98|         this.consent = 'accepted';
99|         this.saveConsent();
100|         this.hideBanner();
101|         this.loadAuthorizedScripts();
102|         // Émettre un événement pour que d'autres scripts puissent réagir
103|         document.dispatchEvent(new CustomEvent('cookie-consent', { detail: { consent: 'accepted' } }));
104|     },
105| 
106|     /**
107|      * Refuse tous les cookies non essentiels
108|      */
109|     declineAll: function () {
110|         this.consent = 'declined';
111|         this.saveConsent();
112|         this.hideBanner();
113|         // Émettre un événement pour que d'autres scripts puissent réagir
114|         document.dispatchEvent(new CustomEvent('cookie-consent', { detail: { consent: 'declined' } }));
115|     },
116| 
117|     /**
118|      * Cache la bannière avec une animation
119|      */
120|     hideBanner: function () {
121|         if (this.banner) {
122|             this.banner.classList.remove('show');
123|             setTimeout(() => {
124|                 if (this.banner) this.banner.remove();
125|             }, 400);
126|         }
127|     },
128| 
129|     /**
130|      * Récupère le consentement stocké dans le cookie
131|      */
132|     getConsent: function () {
133|         const cookies = document.cookie.split('; ');
134|         for (const cookie of cookies) {
135|             const [name, value] = cookie.split('=');
136|             if (name === this.COOKIE_NAME) {
137|                 return value;
138|             }
139|         }
140|         return null;
141|     },
142| 
143|     /**
144|      * Sauvegarde le consentement dans un cookie
145|      */
146|     saveConsent: function () {
147|         const date = new Date();
148|         date.setTime(date.getTime() + this.EXPIRY_DAYS * 24 * 60 * 60 * 1000);
149|         const expires = 'expires=' + date.toUTCString();
150|         document.cookie = `${this.COOKIE_NAME}=${this.consent}; ${expires}; path=/; SameSite=Lax`;
151|     },
152| 
153|     /**
154|      * Charge les scripts qui nécessitent un consentement
155|      * (Google Analytics, etc.)
156|      */
157|     loadAuthorizedScripts: function () {
158|         // Google Analytics est déjà chargé via le script dans index.html
159|         // Cette fonction est un hook pour d'autres scripts futurs
160|         document.dispatchEvent(new CustomEvent('cookie-consent-loaded'));
161|     }
162| };
163| 
164| // Initialiser au chargement du DOM
165| document.addEventListener('DOMContentLoaded', function () {
166|     // Attendre que Translator soit chargé
167|     if (typeof Translator !== 'undefined' && Translator.isLoaded) {
168|         CookieConsent.init();
169|     } else {
170|         // Si Translator n'est pas encore chargé, attendre
171|         const checkTranslator = setInterval(() => {
172|             if (typeof Translator !== 'undefined' && Translator.isLoaded) {
173|                 clearInterval(checkTranslator);
174|                 CookieConsent.init();
175|             }
176|         }, 100);
177|     }
178| });
179| 
180| // Exposer l'objet globalement
181| window.CookieConsent = CookieConsent;
