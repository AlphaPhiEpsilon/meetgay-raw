14| function loadHistoryFromSession() {
15|     try {
16|         const saved = sessionStorage.getItem('meetgay_chat_history');
17|         if (saved) {
18|             conversationHistory = JSON.parse(saved);
19|             // Recréer les dates (optionnel)
20|             for (let contact in conversationHistory) {
21|                 conversationHistory[contact].forEach(msg => {
22|                     if (msg.timestamp) msg.timestamp = new Date(msg.timestamp);
23|                 });
24|             }
25|             console.log('✅ Historique chargé depuis sessionStorage');
26|         } else {
27|             console.log('📭 Aucun historique trouvé dans sessionStorage');
28|         }
29|     } catch (e) {
30|         console.warn('❌ Chargement échoué:', e);
31|     }
32|     updateHistoryDisplay();
33| }

53| function updateHistoryDisplay() {
54|     const historyDiv = document.getElementById('historyList');
55|     if (!historyDiv) return;
56| 
57|     const contacts = Object.keys(conversationHistory);
58|     console.log('🔄 updateHistoryDisplay, contacts:', contacts.length);
59| 
60|     if (contacts.length === 0) {
61|         // TRADUCTION : Texte remplacé par Translator.t()
62|         historyDiv.innerHTML = `<div class="empty-message" style="padding: 20px;">${Translator.t('history.no_conversation')}</div>`;
63|         return;
64|     }
65| 
66|     // Trier par dernier message
67|     contacts.sort((a, b) => {
68|         const lastA = conversationHistory[a][conversationHistory[a].length - 1]?.timestamp || 0;
69|         const lastB = conversationHistory[b][conversationHistory[b].length - 1]?.timestamp || 0;
70|         return new Date(lastB) - new Date(lastA);
71|     });
72| 
73|     historyDiv.innerHTML = '';
74| 
75|     contacts.forEach(contact => {
76|         const messages = conversationHistory[contact];
77|         if (!messages || messages.length === 0) return;
78| 
79|         const lastMessage = messages[messages.length - 1];
80|         const lastMessageText = lastMessage.text.length > 60
81|             ? lastMessage.text.substring(0, 60) + '...'
82|             : lastMessage.text;
83| 
84|         const convDiv = document.createElement('div');
85|         convDiv.className = 'history-conversation';
86|         convDiv.setAttribute('data-contact', contact);
87| 
88|         // TRADUCTION : "Moi" remplacé par Translator.t('history.me')
89|         convDiv.innerHTML = `
90|             <div class="history-conversation-header">
91|                 <span class="history-conversation-name">💬 ${escapeHtml(contact)}</span>
92|                 <span class="history-conversation-date">${formatTime(lastMessage.timestamp)}</span>
93|             </div>
94|             <div class="history-last-message">
95|                 ${escapeHtml(lastMessageText)}
96|             </div>
97|             <div class="history-messages-list">
98|                 ${messages.map(msg => `
99|                     <div class="history-message-item ${msg.type}">
100|                         <div class="history-message-sender">${msg.type === 'sent' ? `📤 ${Translator.t('history.me')}` : `✉️ ${escapeHtml(msg.from)}`}</div>
101|                         <div class="history-message-text">${escapeHtml(msg.text)}</div>
102|                         <div class="history-message-time">${formatTime(msg.timestamp)}</div>
103|                     </div>
104|                 `).join('')}
105|             </div>
106|         `;
107| 
108|         const messagesList = convDiv.querySelector('.history-messages-list');
109|         convDiv.addEventListener('click', (e) => {
110|             e.stopPropagation();
111|             messagesList.classList.toggle('open');
112|         });
113| 
114|         historyDiv.appendChild(convDiv);
115|     });
116| }

118| function formatTime(date) {
119|     if (!date) return '';
120|     const d = new Date(date);
121|     return `${d.getHours().toString().padStart(2, '0')}:${d.getMinutes().toString().padStart(2, '0')}`;
122| }
123| 
124| function escapeHtml(text) {
125|     if (!text) return '';
126|     const div = document.createElement('div');
127|     div.textContent = text;
128|     return div.innerHTML;
129| }
