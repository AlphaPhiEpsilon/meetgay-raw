790| // Fonction pour charger les paramètres
791| async function loadSettings() {
792|     const container = document.getElementById('configForm');
793|     if (!container) return;
794| 
795|     // Si le formulaire est déjà chargé, on ne fait rien
796|     if (configLoaded) {
797|         return;
798|     }
799| 
800|     // Marquer comme chargé AVANT de faire quoi que ce soit
801|     configLoaded = true;
802| 
803|     try {
804|         const response = await fetch('/api/admin/settings', {
805|             method: 'GET',
806|             headers: { 'Content-Type': 'application/json' }
807|         });
808| 
809|         if (!response.ok) {
810|             throw new Error(`Erreur HTTP ${response.status}`);
811|         }
812| 
813|         const data = await response.json();
814| 
815|         if (data.success) {
816|             currentSettings = data.settings;
817|             afficherFormulaire(currentSettings);
818|         } else {
819|             container.innerHTML = `<div style="color:red">❌ Erreur: ${data.error || 'Impossible de charger les paramètres'}</div>`;
820|         }
821|     } catch (error) {
822|         console.error('Erreur loadSettings:', error);
823|         container.innerHTML = `<div style="color:red">❌ Erreur réseau: ${error.message}</div>`;
824|     }
825| }

930|         const msg = document.getElementById('msgConfig');
931|         msg.textContent = '⏳ Sauvegarde en cours...';
932|         msg.className = 'config-message loading';

935|         fetch('/api/admin/settings/update-multiple', {
936|             method: 'POST',
937|             headers: { 'Content-Type': 'application/json' },
938|             body: JSON.stringify({ settings: { [key]: value } })
939|         })
940|             .then(response => response.json())
941|             .then(data => {
942|                 if (data.success) {
943|                     msg.textContent = '✅ Sauvegardé';
944|                     msg.className = 'config-message success';
945|                     // Mettre à jour currentSettings
946|                     currentSettings[key] = value;

948|                     msg.textContent = '❌ Erreur: ' + (data.error || 'Inconnue');
949|                     msg.className = 'config-message error';

951|                 // Effacer le message après 3 secondes
952|                 setTimeout(() => {
953|                     msg.className = 'config-message';
954|                     msg.textContent = '';
955|                 }, 3000);
956|             })
957|             .catch(err => {
958|                 msg.textContent = '❌ Erreur réseau';
959|                 msg.className = 'config-message error';
960|                 setTimeout(() => {
961|                     msg.className = 'config-message';
