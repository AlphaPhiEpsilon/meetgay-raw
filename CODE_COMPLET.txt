245| // ========== 4. FILTRES ET RECHERCHE ==========
246| function applyFilters() {
247|     const searchTerm = searchInput.value.toLowerCase();
248|     const role = roleFilter.value;
249|     const status = statusFilter.value;
250| 
251|     filteredUsers = allUsers.filter(u => {
252|         if (searchTerm && !u.pseudo.toLowerCase().includes(searchTerm)) return false;
253| 
254|         if (role !== 'all') {
255|             if (role === 'admin' && u.role !== 'admin') return false;
256|             if (role === 'modo' && u.role !== 'modo') return false;
257|             if (role === 'user' && u.role !== 'user') return false;
258|             if (role === 'guest' && (u.role && u.role !== 'guest')) return false;
259|         }
260| 
261|         if (status !== 'all') {
262|             if (status === 'banned' && !u.is_banned) return false;
263|             if (status === 'warned' && (!u.warnings || u.warnings < 3)) return false;
264|             if (status === 'active' && (u.is_banned || (u.warnings || 0) >= 3)) return false;
265|         }
266| 
267|         return true;
268|     });
269| 
270|     renderUsersTable();
271| 
272| 
273| }
